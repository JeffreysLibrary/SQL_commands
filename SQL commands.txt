==ORDER BY filter
SELECT* FROM (
SELECT intake_rsa.*,
ROW_NUMBER() OVER (
PARTITION BY Fullname
ORDER BY Program_Year__c DESC
) AS rn
FROM intake_rsa)
WHERE rn=1;

==Tabulate Row Occurrences
SELECT Secondary_Disability, 
COUNT(*) AS occurrences 
FROM DE
GROUP BY Secondary_Disability
ORDER BY occurrences DESC;

==Tabulate Row Occurrences WHERE
SELECT Fullname, Black_or_African_American, Primary_Disability,
COUNT(*) AS times
FROM DE
WHERE Black_or_African = 1
GROUP BY Primary_Disability
ORDER BY times DESC;

SELECT Primary_Disability_Source_of_Impairment__c,
COUNT(*) AS times
FROM new
WHERE Black_or_African_American_Data__c = 1
GROUP BY Primary_Disability_Source_of_Impairment__c
ORDER BY times DESC;

==SELECT MAX() to force integer
SELECT MAX(CAST(ipeWeeklyHoursWorked__c AS INT))
FROM new;

==SELECT MAX() to keep decimal
SELECT MAX(CAST(exitHourlyWage__c AS DECIMAL(18,2)))
FROM new;

==ORDER BY CAST to force text to numbers
SELECT CAST(ipeHourlyWage AS INT)
FROM de
WHERE ipeHourlyWage > 50 
ORDER BY CAST (ipeHourlyWage AS DECIMAL) DESC;
