==ORDER BY filter
SELECT* FROM (
SELECT intake_rsa.*,
ROW_NUMBER() OVER (
PARTITION BY Fullname
ORDER BY Program_Year__c DESC
) AS rn
FROM intake_rsa)
WHERE rn=1;
